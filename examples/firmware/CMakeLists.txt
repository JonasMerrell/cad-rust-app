cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
project(fw C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
foreach(n blink button adc pwm hello multicore irq arm)
  add_executable(${n} ${n}.c)
  target_link_libraries(${n} pico_stdlib hardware_adc hardware_pwm pico_multicore hardware_irq)
  pico_enable_stdio_uart(${n} 1)
  pico_enable_stdio_usb(${n} 0)
  pico_add_extra_outputs(${n})
endforeach()
