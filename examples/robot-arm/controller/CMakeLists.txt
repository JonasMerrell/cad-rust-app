cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
project(arm_controller C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(arm_controller arm_controller.c)
target_link_libraries(arm_controller pico_stdlib hardware_adc hardware_pwm)
pico_enable_stdio_uart(arm_controller 1)
pico_enable_stdio_usb(arm_controller 0)
pico_add_extra_outputs(arm_controller)
